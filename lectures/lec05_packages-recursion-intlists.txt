IntList:
public class IntList {
    int first;
    IntList rest;

    public IntList(int first, IntList rest) {
        this.first = first;
        this.rest = rest;
    }
    public int get(int i){ //Returns the value at index i
        if(i == 0){
            return this.first;
        }
        return rest.get(i - 1)
    }

    public int getIterative(int i){ //This is an iterative approach with for-loop instead of while-loop
        Intlist current = this;
        for(int n = 0; n < i; i++){
            current  = current.rest;
            if(n == i){
                return current.first;
            }
        }
    }
    public IntList incrementRecursiveNonDestructive(){
        IntList newRest = null;
        if(rest != null){
            newRest = rest.incrementRecursiveNonDestructive();
        }
        return new IntList(first + 1, newRest);
    }


    public int size() { //Returns the size of the list
        if(rest == null){
            return 1;
        }
        return rest.size() + 1;
    }

    public static void main(String[] args){
        IntList myList = new IntList();
        myList.first = 67; //Sets first value as 67

        myList.rest = new IntList(); //This creates a box inside of the IntList myList that is another list
        myList.rest.first = 6; //This sets that new list's first value as 6

        myList.rest.rest = new IntList(); //This creates another box inside of the rest value of the IntList
        myList.rest.rest.first = 7; //This sets that first value of that new IntList as 7
        //In summary, this basically allows you to nest lists
        //This is our own custom class for a list

        // V2

        IntList myList = new IntList(67, null);

        myList.rest = new IntList(6, null);

        myList.rest.rest = new IntList(7, null);

        // V3

        IntList l1 = new IntList(5, new IntList(10, new IntList(15, null)));

        IntList l2 = new IntList(5, null);
        l2.rest = new IntList(10, null);
        l2.rest.rest = new IntList(15, null);

        IntList l3 = new IntList(15, null);
        l3 = new IntList(10, l3);
        l3 = new IntList(5, l3);

    }
}
public class TestIntList(){
    @Test
    public void testIntList(){
        IntList l1 = new IntList(5, new IntList(10, new IntList(15, null)));

        assertThat(l1.size()).isEqualTo(3);

        assertThat(l1.get(0)).isEqualTo(5);
        assertThat(l1.get(1)).isEqualTo(10);

        IntList incremented = l1.incrementRecursiveNonDesctructive;
        assertThat(incremented.get(0)).isEqualTo(6);
        assertThat(incremented.get(0)).isEqualTo(11);
        assertThat(incremented.get(0)).isEqualTo(16);
        assertThat(incremented.get(0)).isEqualTo(5);
        assertThat(l1.get(0)).isEqualTo(5);
    }
}
